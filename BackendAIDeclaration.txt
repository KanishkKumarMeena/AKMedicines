def require_roles(*roles):
    def decorator(view):
        @wraps(view)
        def wrapped(*args, **kwargs):
            if g.current_user_role not in roles:
                abort(403, description="Insufficient permissions.")
            return view(*args, **kwargs)

        return wrapped

    return decorator
